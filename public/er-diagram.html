    USERS {
        int id PK "รหัสประจำตัวผู้ใช้"

    }

    OWNERS {
        int owner_id PK "รหัสผู้ประกอบการ"
        string citizen_id "เลขบัตรประชาชน 13 หลัก"
        string juristic_id "เลขนิติบุคคล"
        string owner_name "ชื่อผู้ประกอบการ / เจ้าของ"
        string phone "เบอร์โทรศัพท์ติดต่อ"
        string office_address "ที่อยู่ตามทะเบียนบ้าน"
        string moo "หมู่ที่"
        string subdistrict "ตำบลดอยงาม"

        string province "จังหวัดเชียงราย"
    }

    BUSINESS_CATEGORIES {
        int category_id PK "รหัสหมวดหมู่"
        string category_name UK "ชื่อประเภทกิจการตามเทศบัญญัติ"
    }

    BUSINESSES {
        int business_id PK "รหัสสถานประกอบการ"
        int owner_id FK "รหัสเจ้าของ"
        int category_id FK "รหัสหมวดหมู่"
        string shop_name "ชื่อสถานประกอบการ / ร้านค้า"
        text operation_type "ลักษณะการดำเนินกิจการ"
        string shop_address "ที่ตั้งร้าน"
        string shop_moo "หมู่ที่ตั้งร้าน"
        string business_area "พื้นที่ประกอบการ (ตร.ม.)"
        string machine_power "กำลังเครื่องจักร"
        string worker_count "จำนวนคนงาน"
        string inspection_status "สถานะการตรวจ"
        string workflow_status "ขั้นตอนงาน (สธ.2 / ตรวจ / อภ.2)"
        date request_date "วันที่ยื่นแบบคำขอ สธ.๒"
        string request_no "เลขที่คำขอ (REQ-xxx)"
    }

    LICENSES {
        int license_id PK "รหัสใบอนุญาต"
        int business_id FK "รหัสสถานประกอบการ"
        decimal fee "ค่าธรรมเนียม (บาท)"
        string receipt_book "เล่มที่ใบเสร็จรับเงิน"
        string receipt_no "เลขที่ใบเสร็จรับเงิน"
        date receipt_date "วันที่ออกใบเสร็จ"
        date expire_date "วันสิ้นอายุ (1 ปี)"
        string status "สถานะ (active/expired)"
        boolean is_printed "สถานะพิมพ์ใบ อภ.๒"
        timestamp printed_at "วันเวลาที่พิมพ์"
    }

    INSPECTION_ROUNDS {
        int round_id PK "รหัสรอบตรวจ"
        string round_name UK "ชื่อรอบการออกตรวจ"
        date inspection_date "วันที่ลงพื้นที่ตรวจ"
    }

    INSPECTION_RECORDS {
        int record_id PK "รหัสบันทึกผลตรวจ"
        int round_id FK "รหัสรอบการออกตรวจ"
        int business_id FK "รหัสสถานประกอบการ"
        text condition_4_3 "เงื่อนไขข้อ 4.3"
        text condition_4_4 "เงื่อนไขข้อ 4.4"
        text file_url "ลิงก์ภาพแนบ Google Drive"
        text notes "หมายเหตุและผู้ตรวจ"
    }
สัญลักษณ์และความสัมพันธ์ (Notations)
PK: Primary Key (รหัสหลัก)
FK: Foreign Key (รหัสเชื่อมโยง)
UK: Unique Key (ห้ามซ้ำ)
